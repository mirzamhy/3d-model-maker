import { BufferGeometry, Color, Material, Quaternion, Vector3 } from 'three';
import { hash } from '../hash';
import { fbm, gauss } from '../noise';
import { MeshBuilder, loft, smoothStations, weigher, rigid, ellipsoid, box, fromFrontSigned, type Station, type Weigher } from '../mesh';
import { buildSkeleton, type Skel } from '../skeleton';
import { REAL_BONE_DEFS, REAL_HEAD_Y } from '../skeletonReal';
import { makeVariant, skinAt, clothAt, bloodRun, type RCtx } from '../real/variant';
import { realMaterial } from '../real/materials';

const v = (x: number, y: number, z: number) => new Vector3(x, y, z);
const HY = REAL_HEAD_Y;

function buildHead(ctx: RCtx) {
  const { mb, skel } = ctx;
  mb.use('skin');
  const HR = 1.07;
  
  const keys: Station[] = [
    { c: v(0, HY - 0.006, 0.008), rs: 0.036 * HR, rf: 0.034 * HR },
    { c: v(0, HY + 0.022, 0.034), rs: 0.054 * HR, rf: 0.054 * HR },
    { c: v(0, HY + 0.075, 0.012), rs: 0.070 * HR, rf: 0.083 * HR },
    { c: v(0, HY + 0.150, 0), rs: 0.082 * HR, rf: 0.098 * HR },
    { c: v(0, HY + 0.170, -0.004), rs: 0.084 * HR, rf: 0.101 * HR },
    { c: v(0, HY + 0.228, -0.016), rs: 0.062 * HR, rf: 0.079 * HR },
    { c: v(0, HY + 0.243, -0.017), rs: 0.002 * HR, rf: 0.002 * HR },
  ];
  const st = smoothStations(keys, 3);
  
  const faceFeature = (h: number, a: number) => {
    let out = 0;
    out += -0.030 * gauss(h, 0.148, 0.020) * gauss(Math.abs(a), 0.47, 0.20); // eyes
    out += +0.009 * gauss(h, 0.172, 0.009) * gauss(Math.abs(a), 0.42, 0.32); // brow
    out += +0.010 * gauss(h, 0.145, 0.014) * gauss(Math.abs(a), 0, 0.09); // nose bridge
    out += -0.020 * gauss(h, 0.118, 0.012) * gauss(Math.abs(a), 0, 0.14); // nostril
    out += +0.012 * gauss(h, 0.128, 0.013) * gauss(Math.abs(a), 0.82, 0.18); // cheekbone
    out += -0.020 * gauss(h, 0.092, 0.022) * gauss(Math.abs(a), 0.74, 0.22); // hollow cheek
    out += -0.010 * gauss(h, 0.175, 0.022) * gauss(Math.abs(a), 1.28, 0.20); // temple
    out += -0.010 * gauss(h, 0.085, 0.016) * gauss(Math.abs(a), 0, 0.45); // mouth
    out += +0.009 * gauss(h, 0.028, 0.015) * gauss(Math.abs(a), 0, 0.25); // chin
    out += +0.007 * gauss(h, 0.050, 0.016) * gauss(Math.abs(a), 1.15, 0.25); // jaw corner
    out += +0.004 * gauss(h, 0.205, 0.030) * gauss(Math.abs(a), Math.PI, 0.60); // back head
    return out;
  };

  const MOUTH_SPLIT = 0.084;
  const wHead = rigid(skel, 'Head');
  const wJaw = rigid(skel, 'Jaw');

  loft(mb, st, {
    sides: 32,
    front: v(0, 0, 1),
    weigh: (p) => (p.y < HY + MOUTH_SPLIT && p.z > 0.012 ? wJaw(p) : wHead(p)),
    capEnd: 'flat',
    shape: (i, phi, c) => faceFeature(c.y - HY, fromFrontSigned(phi)),
    paint: {
      base: new Color(),
      rule: (p) => {
        const h = p.y - HY;
        const a = fromFrontSigned(Math.atan2(p.x, p.z));
        const depth = faceFeature(h, a);
        const dark = depth < 0 ? Math.min(1, Math.abs(depth) / 0.022) * 0.72 : 0;
        let red = 0;
        if (h > 0.06 && h < 0.13 && Math.abs(a) < 0.5) red = 0.5; // around mouth
        if (h > 0.10 && h < 0.13 && Math.abs(a) < 0.2) red = 0.8; // nose hole
        if (h > 0.079 && h < 0.095 && Math.abs(a) < 0.62) return new Color('#1c0605'); // mouth band
        return skinAt(ctx, p, dark, red);
      }
    }
  });

  // Eyes
  mb.use('eye');
  for (const s of [1, -1]) {
    ellipsoid(mb, v(s * 0.031, HY + 0.148, 0.075 - 0.031), v(0.0135, 0.0135, 0.0135), wHead, { base: new Color('#c4d8dc') });
    ellipsoid(mb, v(s * 0.031, HY + 0.148, 0.075 - 0.031 + 0.013), v(0.003, 0.003, 0.001), wHead, { base: new Color('#333') }); // pupil
  }

  // Mouth cavity
  mb.use('skin');
  ellipsoid(mb, v(0, HY + 0.08, 0.03), v(0.04, 0.03, 0.04), wHead, { base: new Color('#160404') }, 2);

  // Teeth
  mb.use('teeth');
  for (let top = 0; top <= 1; top++) {
    const y = HY + (top ? 0.098 : 0.071);
    const n = top ? 12 : 11;
    const w = top ? wHead : wJaw;
    for (let i = 0; i < n; i++) {
      if (hash(ctx.seed, 100 + top * 20 + i) < 0.1) continue;
      const a = (i - (n - 1) / 2) * 0.12;
      const p = v(Math.sin(a) * 0.035, y, Math.cos(a) * 0.045 - 0.011);
      const c = new Color('#c4b27c').lerp(new Color('#6e5a32'), hash(ctx.seed, 200 + i));
      box(mb, p, v(0.0036, 0.006, 0.002), w, { base: c }, v(Math.cos(a), 0, -Math.sin(a)));
    }
  }

  // Ears
  mb.use('skin');
  for (const s of [1, -1]) {
    const q = new Quaternion().setFromAxisAngle(v(0, 1, 0), s * 0.25);
    ellipsoid(mb, v(s * 0.08, HY + 0.14, -0.01), v(0.006, 0.025, 0.015), wHead, { base: skinAt(ctx, v(s * 0.08, HY + 0.14, -0.01)) }, 1, q);
  }
}

function buildBody(ctx: RCtx) {
  const { mb, skel, V } = ctx;
  mb.use('skin');
  
  // Neck
  const wNeck = weigher(skel, ['Spine2', 'Neck', 'Head']);
  loft(mb, smoothStations([
    { c: v(0, 1.4, -0.02), rs: 0.06 * V.bulk, rf: 0.05 * V.bulk },
    { c: v(0, HY + 0.01, -0.01), rs: 0.04 * V.bulk, rf: 0.045 * V.bulk }
  ], 3), {
    sides: 16, front: v(0, 0, 1), weigh: wNeck,
    shape: (i, phi) => {
      const a = fromFrontSigned(phi);
      return (Math.abs(a) < 0.2 ? 0.005 : 0) + (Math.abs(a) > 1.2 ? 0.003 : 0);
    },
    paint: { base: new Color(), rule: (p) => skinAt(ctx, p) }
  });

  // Chest / Torso core
  const wSpine = weigher(skel, ['Hips', 'Spine', 'Spine1', 'Spine2']);
  loft(mb, smoothStations([
    { c: v(0, 1.1, -0.01), rs: 0.14 * V.bulk, rf: 0.1 * V.bulk },
    { c: v(0, 1.35, -0.01), rs: 0.16 * V.bulk, rf: 0.12 * V.bulk },
    { c: v(0, 1.45, -0.02), rs: 0.15 * V.bulk, rf: 0.1 * V.bulk },
  ], 3), {
    sides: 24, front: v(0, 0, 1), weigh: wSpine,
    paint: { 
      base: new Color(), 
      rule: (p) => {
        const c = skinAt(ctx, p);
        const run = bloodRun(ctx, p, 0, 0.08, 1.45, 1.1);
        if (run > 0) c.lerp(V.bloodDark, run);
        return c;
      } 
    }
  });

  // Arms and Hands
  for (const s of [1, -1]) {
    const P = s > 0 ? 'Left' : 'Right';
    const wArm = weigher(skel, [`${P}Shoulder`, `${P}Arm`, `${P}ForeArm`, `${P}Hand`]);
    
    // Arm
    loft(mb, smoothStations([
      { c: v(s * 0.18, 1.44, -0.02), rs: 0.045 * V.bulk, rf: 0.05 * V.bulk },
      { c: v(s * 0.5, 1.44, -0.04), rs: 0.035 * V.bulk, rf: 0.04 * V.bulk },
      { c: v(s * 0.76, 1.44, -0.03), rs: 0.025 * V.bulk, rf: 0.03 * V.bulk }
    ], 3), {
      sides: 16, front: v(0, 0, 1), weigh: wArm,
      paint: { base: new Color(), rule: (p) => skinAt(ctx, p) }
    });

    // Fingers
    const fingers = ['Index', 'Middle', 'Ring', 'Pinky', 'Thumb'];
    for (const f of fingers) {
      const isThumb = f === 'Thumb';
      const bNames = isThumb 
        ? [`${P}Hand`, `${P}HandThumb1`, `${P}HandThumb2`, `${P}HandThumb3`]
        : [`${P}Hand`, `${P}Hand${f}1`, `${P}Hand${f}2`, `${P}Hand${f}3`];
      
      const wFinger = weigher(skel, bNames);
      const z0 = isThumb ? 0.03 : (f === 'Index' ? 0.031 : f === 'Middle' ? 0.009 : f === 'Ring' ? -0.013 : -0.033);
      const x0 = isThumb ? 0.782 : (f === 'Pinky' ? 0.852 : 0.86);
      const len = isThumb ? 0.08 : 0.1;
      
      loft(mb, [
        { c: v(s * x0, 1.44, -0.03 + z0), rs: 0.008, rf: 0.008 },
        { c: v(s * (x0 + len), 1.44, -0.03 + z0), rs: 0.006, rf: 0.006 }
      ], {
        sides: 8, front: v(0, 1, 0), weigh: wFinger, capEnd: 'flat',
        paint: { 
          base: new Color(), 
          rule: p => skinAt(ctx, p, 0, Math.abs(p.x) > x0 + 0.06 ? 0.8 : 0) // Bloody tips
        }
      });
      
      // Nails
      mb.use('teeth'); // Using teeth bucket for yellow nails
      box(
        mb, 
        v(s * (x0 + len - 0.005), 1.446, -0.03 + z0), 
        v(0.006, 0.001, 0.005), 
        wFinger, 
        { base: new Color('#7a6f58') }
      );
      mb.use('skin');
    }
  }
}

function buildClothes(ctx: RCtx) {
  const { mb, skel, V } = ctx;
  
  // T-Shirt
  mb.use('shirt');
  const wShirt = weigher(skel, ['Hips', 'Spine', 'Spine1', 'Spine2', 'Neck']);
  loft(mb, smoothStations([
    { c: v(0, 1.05, -0.01), rs: 0.145 * V.bulk, rf: 0.105 * V.bulk },
    { c: v(0, 1.35, -0.01), rs: 0.165 * V.bulk, rf: 0.125 * V.bulk },
    { c: v(0, 1.48, -0.02), rs: 0.155 * V.bulk, rf: 0.105 * V.bulk }
  ], 6), {
    sides: 48, front: v(0, 0, 1), weigh: wShirt,
    open: (phi, ring, c) => c.y > 1.35 && Math.abs(fromFrontSigned(phi)) < ((c.y - 1.35) / 0.15) * 0.5,
    paint: { 
      base: new Color(), 
      rule: p => {
        const c = clothAt(ctx, V.shirt, p);
        const run = bloodRun(ctx, p, 0, 0.12, 1.45, 1.1);
        if (run > 0) c.lerp(V.blood, run);
        return c;
      } 
    }
  });

  // Jacket
  mb.use('jacket');
  const wJacket = weigher(skel, ['Hips', 'Spine', 'Spine1', 'Spine2', 'LeftShoulder', 'RightShoulder', 'Neck']);
  loft(mb, smoothStations([
    { c: v(0, 0.85, -0.01), rs: 0.17 * V.bulk, rf: 0.135 * V.bulk },
    { c: v(0, 1.2, -0.01), rs: 0.18 * V.bulk, rf: 0.145 * V.bulk },
    { c: v(0, 1.52, -0.03), rs: 0.18 * V.bulk, rf: 0.13 * V.bulk }
  ], 3), {
    sides: 40, front: v(0, 0, 1), weigh: wJacket,
    open: (phi, ring, c) => Math.abs(fromFrontSigned(phi)) < 0.2 + 0.32 * (ring / 12), // Open front widening downwards
    shape: (i, phi, c) => 0.005 * Math.sin(phi * 8), // Vertical folds
    paint: { base: new Color(), rule: p => clothAt(ctx, V.jacket, p) }
  });

  // Pants
  mb.use('pants');
  const wPants = weigher(skel, ['Hips', 'Spine']);
  loft(mb, smoothStations([
    { c: v(0, 0.98, 0), rs: 0.16 * V.bulk, rf: 0.12 * V.bulk },
    { c: v(0, 0.88, 0), rs: 0.17 * V.bulk, rf: 0.13 * V.bulk }
  ], 2), {
    sides: 32, front: v(0, 0, 1), weigh: wPants,
    paint: { base: new Color(), rule: p => clothAt(ctx, V.pants, p) }
  });
  
  for (const s of [1, -1]) {
    const P = s > 0 ? 'Left' : 'Right';
    
    // Jacket Sleeve
    mb.use('jacket');
    const wSleeve = weigher(skel, [`${P}Shoulder`, `${P}Arm`, `${P}ForeArm`]);
    loft(mb, smoothStations([
      { c: v(s * 0.18, 1.44, -0.02), rs: 0.06 * V.bulk, rf: 0.065 * V.bulk },
      { c: v(s * 0.5, 1.44, -0.04), rs: 0.05 * V.bulk, rf: 0.055 * V.bulk },
      { c: v(s * 0.72, 1.44, -0.03), rs: 0.04 * V.bulk, rf: 0.045 * V.bulk }
    ], 3), {
      sides: 20, front: v(0, 0, 1), weigh: wSleeve,
      shape: (i, phi, c) => 0.004 * Math.sin(phi * 6), // Sleeve wrinkles
      paint: { base: new Color(), rule: p => clothAt(ctx, V.jacket, p) }
    });

    // Pant Leg
    mb.use('pants');
    const wLeg = weigher(skel, ['Hips', `${P}UpLeg`, `${P}Leg`, `${P}Foot`]);
    loft(mb, smoothStations([
      { c: v(s * 0.1, 0.95, 0), rs: 0.09 * V.bulk, rf: 0.1 * V.bulk },
      { c: v(s * 0.11, 0.52, 0.015), rs: 0.07 * V.bulk, rf: 0.08 * V.bulk },
      { c: v(s * 0.115, 0.12, -0.01), rs: 0.065 * V.bulk, rf: 0.075 * V.bulk }
    ], 3), {
      sides: 24, front: v(0, 0, 1), weigh: wLeg,
      shape: (i, phi, c) => (c.y < 0.6 ? 0.005 * Math.sin(phi * 5) : 0), // Knee/bottom folds
      paint: { base: new Color(), rule: p => clothAt(ctx, V.pants, p) }
    });

    // Leather Shoe
    mb.use('leather');
    const wShoe = weigher(skel, [`${P}Foot`, `${P}ToeBase`, `${P}Toe_End`]);
    loft(mb, [
      { c: v(s * 0.115, 0.1, -0.05), rs: 0.045, rf: 0.05 },
      { c: v(s * 0.115, 0.03, 0.1), rs: 0.052, rf: 0.065 },
      { c: v(s * 0.115, 0.025, 0.23), rs: 0.048, rf: 0.02 } // Intentional wider shoe per notes
    ], {
      sides: 20, front: v(0, 1, 0), weigh: wShoe, capEnd: 'flat', capStart: 'flat',
      paint: { base: V.shoe }
    });
  }

  // Belt
  mb.use('leather');
  loft(mb, [
    { c: v(0, 0.99, 0), rs: 0.162 * V.bulk, rf: 0.122 * V.bulk },
    { c: v(0, 0.95, 0), rs: 0.162 * V.bulk, rf: 0.122 * V.bulk }
  ], {
    sides: 40, front: v(0, 0, 1), weigh: wPants,
    paint: { base: V.belt }
  });

  // Belt Buckle
  mb.use('metal');
  box(mb, v(0, 0.97, 0.125 * V.bulk), v(0.025, 0.02, 0.005), wPants, { base: new Color('#a89f91') });
}

export function buildArgon(seed: number): { skel: Skel; mb: MeshBuilder; geometry: BufferGeometry; material: Material[] } {
  const skel = buildSkeleton(REAL_BONE_DEFS);
  const mb = new MeshBuilder(seed, true);
  const V = makeVariant(seed);
  const ctx: RCtx = { skel, mb, seed, V };

  buildHead(ctx);
  buildBody(ctx);
  buildClothes(ctx);

  const { geometry, materials } = mb.geometry();
  const material = materials.map((n) => realMaterial(n));
  
  return { skel, mb, geometry, material };
}
